cmake_minimum_required(VERSION 3.25)
include(CMakeOptions.cmake)
project(FileSearch VERSION 0.1.0 LANGUAGES C)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set_property(GLOBAL PROPERTY USE_FOLDERS ON)
if(WIN32)
  set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/bin")
endif()

include(cmake/CmakeUtils.cmake)
include(cmake/Sanitizers.cmake)
include(cmake/FsxDependencies.cmake)
if(FSX_BUILD_PARSER_TOOLS)
  add_subdirectory(tools)
  include(cmake/FindTools.cmake)
endif()
if(FSX_BUILD_NTFS_LIVE_TESTS AND NOT BUILD_TESTING)
  message(FATAL_ERROR "FSX_BUILD_NTFS_LIVE_TESTS requires BUILD_TESTING=ON")
endif()
if(BUILD_TESTING)
  include(CTest)
endif()

if(NOT FSX_BUILD_PARSER_TOOLS)
  message(FATAL_ERROR "The query lexer requires FSX_BUILD_PARSER_TOOLS=ON")
endif()
add_subdirectory(src)
if(BUILD_TESTING)
  add_subdirectory(tests)
endif()
